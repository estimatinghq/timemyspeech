  <script>
    function showPanel(panelName, button) {
      document.querySelectorAll(".panel").forEach(panel => panel.classList.remove("active"));
      document.querySelectorAll(".tab-btn").forEach(btn => btn.classList.remove("active"));

      if (panelName === "password") {
        document.getElementById("passwordPanel").classList.add("active");
      } else if (panelName === "number") {
        document.getElementById("numberPanel").classList.add("active");
      } else if (panelName === "coin") {
        document.getElementById("coinPanel").classList.add("active");
      }

      button.classList.add("active");
    }

    function generatePassword() {
      const length = parseInt(document.getElementById("passwordLength").value, 10);
      const includeUppercase = document.getElementById("includeUppercase").checked;
      const includeLowercase = document.getElementById("includeLowercase").checked;
      const includeNumbers = document.getElementById("includeNumbers").checked;
      const includeSymbols = document.getElementById("includeSymbols").checked;
      const result = document.getElementById("passwordResult");
      const helper = document.getElementById("passwordHelper");

      const uppercase = "ABCDEFGHIJKLMNOPQRSTUVWXYZ";
      const lowercase = "abcdefghijklmnopqrstuvwxyz";
      const numbers = "0123456789";
      const symbols = "!@#$%^&*()_+-=[]{}|;:,.<>?";

      let chars = "";

      if (includeUppercase) chars += uppercase;
      if (includeLowercase) chars += lowercase;
      if (includeNumbers) chars += numbers;
      if (includeSymbols) chars += symbols;

      if (!chars) {
        result.textContent = "Select at least one option";
        helper.textContent = "";
        return;
      }

      const safeLength = Math.max(4, Math.min(64, length || 12));
      let password = "";

      for (let i = 0; i < safeLength; i++) {
        const randomIndex = Math.floor(Math.random() * chars.length);
        password += chars[randomIndex];
      }

      result.textContent = password;
      helper.textContent = `Generated ${safeLength} character password`;
    }

    function generateRandomNumber() {
      const min = parseInt(document.getElementById("minNumber").value, 10);
      const max = parseInt(document.getElementById("maxNumber").value, 10);
      const result = document.getElementById("numberResult");
      const helper = document.getElementById("numberHelper");

      if (isNaN(min) || isNaN(max)) {
        result.textContent = "Enter valid numbers";
        helper.textContent = "";
        return;
      }

      if (min > max) {
        result.textContent = "Minimum cannot be greater than maximum";
        helper.textContent = "";
        return;
      }

      const randomNumber = Math.floor(Math.random() * (max - min + 1)) + min;
      result.textContent = randomNumber;
      helper.textContent = `Generated number between ${min} and ${max}`;
    }

    function flipCoin() {
      const result = document.getElementById("coinResult");
      const helper = document.getElementById("coinHelper");

      const coin = Math.random() < 0.5 ? "Heads" : "Tails";
      result.textContent = coin;
      helper.textContent = "Random coin flip complete";
    }

    function copyResult(resultId, buttonId, originalText) {
      const resultText = document.getElementById(resultId).textContent;
      const button = document.getElementById(buttonId);

      if (
        resultText === "Your password will appear here" ||
        resultText === "Your number will appear here" ||
        resultText === "Heads or Tails?" ||
        resultText === "Select at least one option" ||
        resultText === "Enter valid numbers" ||
        resultText === "Minimum cannot be greater than maximum"
      ) {
        return;
      }

      navigator.clipboard.writeText(resultText);

      button.textContent = "Copied";
      button.style.background = "#22c55e";
      button.style.color = "white";
      button.style.border = "none";

      setTimeout(() => {
        button.textContent = originalText;
        button.style.background = "#f3f4f6";
        button.style.color = "#111827";
        button.style.border = "1px solid #d1d5db";
      }, 1500);
    }

    generatePassword();
    generateRandomNumber();
  </script>
